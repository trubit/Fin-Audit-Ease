import type { AuditRequest, CreateAuditRequestPayload } from '../types'

export const auditRequestService = {
  /**
   * Create a new auditor Provided by Client (PBC) request.
   * Transmits POST /api/v1/audit-requests
   */
  async createAuditRequest(payload: CreateAuditRequestPayload): Promise<ApiResponse<AuditRequest>> {
    return apiClient.post<AuditRequest>(
      API_ENDPOINTS.AUDIT_REQUESTS.CREATE, 
      payload
    )
  },
}

export async function createAuditRequest(req: Request, res: Response) {
  // 1. Authenticate user from JWT context
  const { userId, role, organizationId } = req.user;
  if (role !== 'AUDITOR' && role !== 'AUDIT_MANAGER') {
    return res.status(403).json({ success: false, error: { code: 'FORBIDDEN' } });
  }

  // 2. Enforce Tenant Isolation
  const engagement = await db.findEngagement(req.body.engagementId);
  if (engagement.auditorOrgId !== organizationId) {
    return res.status(403).json({ success: false, error: { code: 'TENANT_VIOLATION' } });
  }

  // 3. Insert Database Record & Return Standard Envelope
  const newRequest = await db.insertAuditRequest(req.body, userId);
  return res.status(201).json({
    success: true,
    data: newRequest
  });
}
Separate Administration Platform Security Gateway Connection ADMIN SEPARATION GUARANTEE

Main Client Application (app.finauditease.com)

• Base Route: /login, /signup
• Roles Allowed: AUDITOR, FINANCE_USER, AUDIT_MANAGER
• State Storage: fae_dev_auth_session (Zustand useAuthStore)
• JWT Audience: aud: "main-app"
• Guard: ProtectedRoute.tsx

Separate Admin Platform (admin.finauditease.com)

• Base Route: /admin/login (NO public signup)
• Roles Allowed: Strictly ADMIN
• State Storage: fae_admin_session (Zustand useAdminAuthStore)
• JWT Audience: aud: "admin-platform" (Separate Token Pipeline)
• Guard: AdminProtectedRoute.tsx

Multi-Tenant Database Isolation & Security Layer BACKEND RESPONSIBILITY

Relational PostgreSQL Schema with Tenant Foreign Keys

• users: id, email, password_hash, role, organization_id, is_active
• organizations: id, name, type (AUDIT_FIRM|CORPORATE), subscription_tier
• engagements: id, title, auditor_org_id, client_org_id, status, risk_score
• audit_requests: id, engagement_id, title, category, status, due_date
• documents: id, request_id, file_name, s3_key, sha256_hash, file_size
• audit_trail: id, actor_id, org_id, action, resource, timestamp, ip_address

Mandatory Security Guardrails

• Row-Level Security (RLS): Enforce WHERE org_id = current_setting('app.current_org_id')
• Encryption at Rest: AES-256 for database tables and S3/GCS buckets
• Audit Immutability: Append-only log for all compliance actions
• Rate Limiting: Enforce 100 req/min for standard APIs and 5 req/min for auth endpoints
